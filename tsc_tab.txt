local colCache = {}
local function colorFor(name)
    local c = colCache[name]
    if c then return c end
    local u = string.upper(name)
    local has = function(s) return string.find(u, s, 1, true) ~= nil end
    local ends = function(s) return #u >= #s and string.sub(u, -#s) == s end
    if GUNS[name] then c = COL.gun
    elseif has("VIAL") then c = COL.vial
    elseif has("NUCLEAR AUTHENTICATION") then c = COL.key
    elseif ends(" KEY") or ends(" KEYS") then c = COL.key
    else c = COL.other end
    colCache[name] = c
    return c
end

local owned = {}
local function take(kind)
    STASH[kind] = STASH[kind] or {}
    local list = STASH[kind]
    local d = list[#list]
    if d then
        list[#list] = nil
    else
        d = Drawing.new(kind)
        if kind == "Text" and STASH.defFont == nil then
            local ok, f = pcall(function() return d.Font end)
            STASH.defFont = ok and f or false
        end
    end
    d.Visible = false
    owned[#owned + 1] = { kind = kind, d = d }
    return d
end

            lab.name.Color = col.text

local function addZone(out, part, name, col)
    if not part or part.ClassName ~= "Part" then return end
    local cf, sz = part.CFrame, part.Size
    local c = cf.Position
    local ax = { cf.RightVector, cf.UpVector, cf.LookVector }
    local h = { sz.X / 2, sz.Y / 2, sz.Z / 2 }
    local corners = {}
    for k = 0, 7 do
        local sx = (k % 2 == 1) and 1 or -1
        local sy = (math.floor(k / 2) % 2 == 1) and 1 or -1
        local sw = (math.floor(k / 4) % 2 == 1) and 1 or -1
        corners[k + 1] = c + ax[1] * (h[1] * sx) + ax[2] * (h[2] * sy) + ax[3] * (h[3] * sw)
    end
    out[#out + 1] = { name = name, col = col, c = c, ax = ax, h = h, corners = corners, top = c + ax[2] * h[2] }
end

local CX, CY, CZ, SP = {}, {}, {}, {}
local zlast, zshown = 0, 0
RunService.RenderStepped:Connect(function()
    if not running then return end
    local clk = os.clock()
    if clk - zlast < ZRATE then return end
    zlast = clk
    local used = 0
    local cam = zcam
    if S.zones and cam and #zones > 0 then
        local cf = cam.CFrame
        local vs = cam.ViewportSize
        local cp, r, u, l = cf.Position, cf.RightVector, cf.UpVector, cf.LookVector
        local hw, hh = vs.X / 2, vs.Y / 2
        local k = hh / math.tan(math.rad(cam.FieldOfView) / 2)
        local me = zroot and zroot.Position
        for _, z in ipairs(zones) do
            if used >= MAXZ then break end
            local inside, dist = false, 0
            if me then
                local rel = me - z.c
                local ex = 0
                for a = 1, 3 do
                    local o = math.abs(rel:Dot(z.ax[a])) - z.h[a]
                    if o > 0 then ex = ex + o * o end
                end
                dist = math.sqrt(ex)
                inside = ex == 0
            end
            if dist <= ZDIST then
                used = used + 1
                local s = zpool[used]
                local col = z.col
                if s.col ~= col then
                    for e = 1, 12 do s.lines[e].Color = col.dot end
                    for f = 1, 6 do s.quads[f].Color = col.dot end
                    s.lab.name.Color = col.text
                    s.lab.dist.Color = col.sub
                    s.col = col
                end
                for i = 1, 8 do
                    local rel = z.corners[i] - cp
                    local zz = rel:Dot(l)
                    CX[i], CY[i], CZ[i] = rel:Dot(r), rel:Dot(u), zz
                    SP[i] = zz >= NEAR and Vector2.new(hw + CX[i] / zz * k, hh - CY[i] / zz * k) or nil
                end
                for e = 1, 12 do
                    local a, b = EDGES[e][1], EDGES[e][2]
                    local za, zb = CZ[a], CZ[b]
                    local on = false
                    if za >= NEAR or zb >= NEAR then
                        local pa, pb = SP[a], SP[b]
                        if not pa or not pb then
                            local f, x, y
                            if not pa then
                                f = (NEAR - za) / (zb - za)
                                x, y = CX[a] + (CX[b] - CX[a]) * f, CY[a] + (CY[b] - CY[a]) * f
                                pa = Vector2.new(hw + x / NEAR * k, hh - y / NEAR * k)
                            else
                                f = (NEAR - zb) / (za - zb)
                                x, y = CX[b] + (CX[a] - CX[b]) * f, CY[b] + (CY[a] - CY[b]) * f
                                pb = Vector2.new(hw + x / NEAR * k, hh - y / NEAR * k)
                            end
                        end
                        local ln = s.lines[e]
                        ln.From = pa; ln.To = pb
                        on = true
                    end
                    if s.lon[e] ~= on then s.lines[e].Visible = on; s.lon[e] = on end
                end
                for f = 1, 6 do
                    local fa = FACES[f]
                    local n = z.ax[fa[1]] * fa[2]
                    local fc = z.c + n * z.h[fa[1]]
                    local idx = fa[3]
                    local on = false
                    if not inside and (cp - fc):Dot(n) > 0 and SP[idx[1]] and SP[idx[2]] and SP[idx[3]] and SP[idx[4]] then
                        local q = s.quads[f]
                        q.PointA = SP[idx[1]]; q.PointB = SP[idx[2]]; q.PointC = SP[idx[3]]; q.PointD = SP[idx[4]]
                        on = true
                    end
                    if s.qon[f] ~= on then s.quads[f].Visible = on; s.qon[f] = on end
                end
                local showN, showD = false, false
                if S.znames then
                    local rel = z.top - cp
                    local zz = rel:Dot(l)
                    if zz >= NEAR then
                        local x = hw + rel:Dot(r) / zz * k
                        local y = hh - rel:Dot(u) / zz * k
                        setTxt(s.lab, "name", z.name, Vector2.new(x, y - 8 - 2 * LINEH))
                        setTxt(s.lab, "dist", inside and "ur inside" or (math.floor(dist) .. "m"), Vector2.new(x, y - 8 - LINEH))
                        showN, showD = true, true
                    end
                end
                show(s.lab, "name", showN)
                show(s.lab, "dist", showD)
            end
        end
    end
    for i = used + 1, zshown do zhide(zpool[i]) end
    zshown = used
end)

local function inList(csv, name)
    if type(csv) ~= "string" then return false end
    for s in string.gmatch(csv, "[^,]+") do if s == name then return true end end
    return false
end

local function hasTool(char, n)
    local bp = lp:FindFirstChild("Backpack")
    return (bp and bp:FindFirstChild(n)) or (char and char:FindFirstChild(n))
end

local function targetsMe(t, char, eps)
    if not char then return true end
    if char:GetAttribute("ABOMINATION") == "secondary" then return false end
    local team = lp.Team and lp.Team.Name or ""
    local hit = false
    if t:GetAttribute("ShootAll") then
        hit = true
    elseif eps and not t:GetAttribute("EpsilonExempt") then
        hit = not (hasTool(char, "SECURITY-CARD") or hasTool(char, "CTZ-CARD") or hasTool(char, "SPEC-ACCESS"))
    elseif TS_TEAMS[team] then
        local tagged = char:GetAttribute("Tagged")
        hit = not (tagged or char:FindFirstChild("TSAnkleMonitor") or hasTool(char, "ID Card") or hasTool(char, "CIS-FREEROAM-ACCESS"))
    end
    if inList(t:GetAttribute("TargetTeams"), team) then hit = true end
    if inList(t:GetAttribute("SafeTeams"), team) then hit = false end
    return hit
end

                local rad = rr > 0 and math.sqrt(rr) or nil
                if rad then
                    for e = 0, TSEG do
                        local rel = Vector3.new(t.pos.X + RCOS[e] * rad, footY, t.pos.Z + RSIN[e] * rad) - cp
                        local zz = rel:Dot(l)
                        PX[e], PY[e], PZ[e] = rel:Dot(r), rel:Dot(u), zz
                        PS[e] = zz >= NEAR and Vector2.new(hw + PX[e] / zz * k, hh - PY[e] / zz * k) or nil
                    end
                end
                for e = 1, TSEG do
                    local on = false
                    if rad then
                        local a, b = e - 1, e
                        local za, zb = PZ[a], PZ[b]
                        if za >= NEAR or zb >= NEAR then
                            local pa, pb = PS[a], PS[b]
                            if not pa or not pb then
                                local f, x, y
                                if not pa then
                                    f = (NEAR - za) / (zb - za)
                                    x, y = PX[a] + (PX[b] - PX[a]) * f, PY[a] + (PY[b] - PY[a]) * f
                                    pa = Vector2.new(hw + x / NEAR * k, hh - y / NEAR * k)
                                else
                                    f = (NEAR - zb) / (za - zb)
                                    x, y = PX[b] + (PX[a] - PX[b]) * f, PY[b] + (PY[a] - PY[b]) * f
                                    pb = Vector2.new(hw + x / NEAR * k, hh - y / NEAR * k)
                                end
                            end
                            local ln = s.lines[e]
                            ln.From = pa; ln.To = pb
                            on = true
                        end
                    end
                    if s.lon[e] ~= on then s.lines[e].Visible = on; s.lon[e] = on end
                end
                local showN, showD = false, false
                local rel = t.pos - cp
                local zz = rel:Dot(l)
                if zz >= NEAR then
                    local x = hw + rel:Dot(r) / zz * k
                    local y = hh - rel:Dot(u) / zz * k
                    local sub = t.why or (d <= TRANGE and "in range" or (math.floor(d - TRANGE) .. "m"))
                    setTxt(s.lab, "name", "turret", Vector2.new(x, y - 8 - 2 * LINEH))
                    setTxt(s.lab, "dist", sub, Vector2.new(x, y - 8 - LINEH))
                    showN, showD = true, true
                end
                show(s.lab, "name", showN)
                show(s.lab, "dist", showD)
            end
        end
    end
    for i = used + 1, tshown do thide(tpool[i]) end
    tshown = used
end)

local MSK = "TSC_MS_" .. tostring(game.JobId)
local ms = _G[MSK]
local msBusy = false
local msApplied = ms and ms.on or false

local function msWrite(on)
    if not ms then return end
    if on then
        applygc(ms.G, "GridSize", 1)
        applygc(ms.C, "CellSize", 50)
        applygc(ms.M, "Mines", 0)
    else
        for i = 1, #ms.G do
            applygc({ ms.G[i] }, "GridSize", ms.og[i])
            applygc({ ms.C[i] }, "CellSize", ms.oc[i])
            applygc({ ms.M[i] }, "Mines", ms.om[i])
        end
    end
    ms.on = on
    msApplied = on
end

local function scanMines(after)
    if msBusy then return end
    msBusy = true
    task.spawn(function()
        local o = os.clock()
        local hits = getgc({ "GridSize", "CellSize", "Mines" })
        local grid, cell, mines = {}, {}, {}
        for _, e in ipairs(hits) do
            if e.type == "number" then
                if e.key == "GridSize" then grid[#grid + 1] = e
                elseif e.key == "CellSize" then cell[#cell + 1] = e
                elseif e.key == "Mines" then mines[#mines + 1] = e end
            end
        end
        local G, C, M = {}, {}, {}
        for _, g in ipairs(grid) do
            local c, m
            for _, x in ipairs(cell) do if x.addr > g.addr and x.addr - g.addr <= 120 then c = x end end
            for _, x in ipairs(mines) do if x.addr > g.addr and x.addr - g.addr <= 120 then m = x end end
            if c and m then G[#G + 1] = g; C[#C + 1] = c; M[#M + 1] = m end
        end
        if #G ~= 7 then
            notify("TSC", "found " .. #G .. " instead of 7 so im leaving it", 3)
            msBusy = false
            pcall(UI.SetValue, "tsc_mines", false)
            S.mines = false
            return
        end
        local og, oc, om = {}, {}, {}
        local already = true
        for i = 1, #G do
            og[i], oc[i], om[i] = G[i].value, C[i].value, M[i].value
            if not (og[i] == 1 and oc[i] == 50 and om[i] == 0) then already = false end
        end
        if already then
            notify("TSC", "values already look modded, off might not restore. rejoin to fully reset", 4)
        end
        ms = { G = G, C = C, M = M, og = og, oc = oc, om = om, on = false }
        _G[MSK] = ms
        msBusy = false
        notify("TSC", "minesweeper scan took " .. string.format("%.1f", os.clock() - o) .. "s", 3)
        if after then after() end
    end)
end

local function setMines(on)
    if on == msApplied and ms then return end
    if not ms then
        if not on then return end
        scanMines(function() msWrite(S.mines) end)
        return
    end
    msWrite(on)
    notify("TSC", "minesweeper " .. (on and "on" or "off"), 2)
end

S.mines = msApplied
task.spawn(function()
    while running do
        local ok, v = pcall(UI.GetValue, "tsc_mines")
        if ok and v ~= nil then S.mines = v end
        if not msBusy and (S.mines ~= msApplied or (S.mines and not ms)) then
            setMines(S.mines)
        end
        task.wait(0.25)
    end
end)
