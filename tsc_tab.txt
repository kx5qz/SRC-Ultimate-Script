local colCache = {}
local function colorFor(name)
    local c = colCache[name]
    if c then return c end
    local u = string.upper(name)
    local has = function(s) return string.find(u, s, 1, true) ~= nil end
    local ends = function(s) return #u >= #s and string.sub(u, -#s) == s end
    if GUNS[name] then c = COL.gun
    elseif has("VIAL") then c = COL.vial
    elseif has("NUCLEAR AUTHENTICATION") then c = COL.key
    elseif ends(" KEY") or ends(" KEYS") then c = COL.key
    else c = COL.other end
    colCache[name] = c
    return c
end

local owned = {}
local function take(kind)
    STASH[kind] = STASH[kind] or {}
    local list = STASH[kind]
    local d = list[#list]
    if d then
        list[#list] = nil
    else
        d = Drawing.new(kind)
        if kind == "Text" and STASH.defFont == nil then
            local ok, f = pcall(function() return d.Font end)
            STASH.defFont = ok and f or false
        end
    end
    d.Visible = false
    owned[#owned + 1] = { kind = kind, d = d }
    return d
end

local NAK_TOOL, NAD_TOOL = "NUCLEAR AUTHENTICATION KEY", "NUCLEAR AUTHENTICATION DISK"
local function nkFind()
    local nak, nad = {}, {}
    local function add(t, where, pos, me)
        local e = { who = where, pos = pos, me = me }
        if t.Name == NAK_TOOL then nak[#nak + 1] = e elseif t.Name == NAD_TOOL then nad[#nad + 1] = e end
    end
    for _, c in ipairs(workspace:GetChildren()) do
        if c.ClassName == "Tool" and (c.Name == NAK_TOOL or c.Name == NAD_TOOL) then
            local h = c:FindFirstChild("Handle")
            local ok, pos = pcall(function() return h.Position end)
            add(c, "dropped", ok and pos or nil)
        end
    end
    for _, p in ipairs(Players:GetPlayers()) do
        local ch = p.Character
        local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
        local ok, pos = pcall(function() return hrp.Position end)
        pos = ok and pos or nil
        if ch then for _, c in ipairs(ch:GetChildren()) do add(c, p.Name, pos, p == lp) end end
        local bp = p:FindFirstChild("Backpack")
        if bp then for _, c in ipairs(bp:GetChildren()) do add(c, p.Name, pos, p == lp) end end
    end
    return nak, nad
end

local GPARTS = { "TeamDetect", "LaserHitbox" }

local gon = { lasers = false, teslas = false }

local function eachGatePart(kind, fn)
    local gates = {}
    if kind == "lasers" then
        local lg = workspace:FindFirstChild("LaserGates")
        if lg then
            for _, area in ipairs(lg:GetChildren()) do
                for _, g in ipairs(area:GetChildren()) do gates[#gates + 1] = g end
            end
        end
    else
        local tg = workspace:FindFirstChild("TeslaGates")
        local gs = tg and tg:FindFirstChild("Gates")
        if gs then gates = gs:GetChildren() end
    end
    for _, g in ipairs(gates) do
        local ms = g:FindFirstChild("MotionSensor")
        if ms then
            for _, n in ipairs(GPARTS) do
                local p = ms:FindFirstChild(n)
                if p then fn(p) end
            end
        end
    end
end

local function gateOff(p)
    local k = tostring(p.Address)
    if gorig[k] then return end
    local pos = p.Position
    if pos.Y > -40000 then
        gorig[k] = pos
        p.Position = FAR
    end
end

local function gateBack(p)
    local k = tostring(p.Address)
    local o = gorig[k]
    if o then
        p.Position = o
        gorig[k] = nil
    end
end

local function restoreGates(kind)
    pcall(eachGatePart, kind, gateBack)
    gon[kind] = false
end

local vorig = {}

local function eachVent(fn)
    local vf = workspace:FindFirstChild("Vents")
    if not vf then return end
    for _, v in ipairs(vf:GetChildren()) do
        if v:GetAttribute("Difficulty") then
            fn(v)

local function ventFast(v)
    local k = tostring(v.Address)
    local o = vorig[k]
    if not o then
        o = v:GetAttribute("Difficulty")
        vorig[k] = o
    end
    local want = o / VENTSPEED
    if v:GetAttribute("Difficulty") ~= want then v:SetAttribute("Difficulty", want) end
end

local function ventBack(v)
    local k = tostring(v.Address)
    local o = vorig[k]
    if o then
        v:SetAttribute("Difficulty", o)
        vorig[k] = nil
    end
end

local function restoreVents()
    pcall(eachVent, ventBack)
    vorig = {}
    vtouched = false
end

RunService.RenderStepped:Connect(function()
    if not running then return end
    local n = 0
    local best, bestD2, bestIt, bestSp
    if S.esp then
        local mx, my = mouse.X, mouse.Y
        for _, it in ipairs(items) do
            if n >= MAXDRAW then break end
            local sp, v = WorldToScreen(it.pos)
            if v then
                n = n + 1
                local slot = dots[n]
                if slot.col ~= it.col then slot.d.Color = it.col.dot; slot.col = it.col end
                if slot.r ~= DOTR then slot.d.Radius = DOTR; slot.r = DOTR end
                slot.d.Position = sp
                if not slot.on then slot.d.Visible = true; slot.on = true end
                if mx and my then
                    local dx, dy = sp.X - mx, sp.Y - my
                    local d2 = dx * dx + dy * dy
                    if d2 <= HOVER * HOVER and (not bestD2 or d2 < bestD2) then
                        best, bestD2, bestIt, bestSp = slot, d2, it, sp
                    end
                end
            end
        end
    end
    for i = n + 1, MAXDRAW do
        local s = dots[i]
        if s.on then s.d.Visible = false; s.on = false end
    end

    local showName, showDist = false, false
    if best then
        best.d.Radius = DOTR_HOVER; best.r = DOTR_HOVER
        local col = bestIt.col
        if lab.col ~= col then
            lab.name.Color = col.text
            lab.dist.Color = col.sub
            lab.col = col
        end
        local lines = (S.names and 1 or 0) + (S.dist and 1 or 0)
        local y = bestSp.Y - 8 - lines * LINEH
        if S.names then
            setTxt(lab, "name", bestIt.name, Vector2.new(bestSp.X, y))
            y = y + LINEH
            showName = true
        end
        if S.dist then
            setTxt(lab, "dist", bestIt.d .. "m", Vector2.new(bestSp.X, y))
            showDist = true
        end
    end
    show(lab, "name", showName)
    show(lab, "dist", showDist)
end)

local GEO, ngeo = {}, 0
local function addZone(out, part, name, col, pos)
    if not part then return end
    local key = tostring(part.Address)
    local hit = GEO[key]
    if hit and hit.name == name then
        out[#out + 1] = hit
        return hit
    end
    local ok, cf, sz = pcall(function() return part.CFrame, part.Size end)
    if not ok or not sz then return end
    pos = pos or gorig[key]
    local c = pos or cf.Position
    local ax = { cf.RightVector, cf.UpVector, cf.LookVector }
    local h = { sz.X / 2, sz.Y / 2, sz.Z / 2 }
    local corners = {}
    for k = 0, 7 do
        local sx = (k % 2 == 1) and 1 or -1
        local sy = (math.floor(k / 2) % 2 == 1) and 1 or -1
        local sw = (math.floor(k / 4) % 2 == 1) and 1 or -1
        corners[k + 1] = c + ax[1] * (h[1] * sx) + ax[2] * (h[2] * sy) + ax[3] * (h[3] * sw)
    end
    local z = { name = name, col = col, c = c, ax = ax, h = h, corners = corners, top = c + ax[2] * h[2] }
    if not pos and c.Y < -40000 then
        out[#out + 1] = z
        return z
    end
    if ngeo > 3000 then GEO, ngeo = {}, 0 end
    GEO[key] = z
    ngeo = ngeo + 1
    out[#out + 1] = z
    return z
end

local function scanMines(after)
    if msBusy then return end
    msBusy = true
    task.spawn(function()
        local o = os.clock()
        local hits = getgc({ "GridSize", "CellSize", "Mines" })
        local grid, cell, mines = {}, {}, {}
        for _, e in ipairs(hits) do
            if e.type == "number" then
                if e.key == "GridSize" then grid[#grid + 1] = e
                elseif e.key == "CellSize" then cell[#cell + 1] = e
                elseif e.key == "Mines" then mines[#mines + 1] = e end
            end
        end
        local G, C, M = {}, {}, {}
        for _, g in ipairs(grid) do
            local c, m
            for _, x in ipairs(cell) do if x.addr > g.addr and x.addr - g.addr <= 120 then c = x end end
            for _, x in ipairs(mines) do if x.addr > g.addr and x.addr - g.addr <= 120 then m = x end end
            if c and m then G[#G + 1] = g; C[#C + 1] = c; M[#M + 1] = m end
        end
        if #G ~= 7 then
            notify("SRC", "found " .. #G .. " instead of 7 so im leaving it", 3)
            msBusy = false
            pcall(UI.SetValue, "tsc_mines", false)
            S.mines = false
            return
        end
        local og, oc, om = {}, {}, {}
        local already = true
        for i = 1, #G do
            og[i], oc[i], om[i] = G[i].value, C[i].value, M[i].value
            if not (og[i] == 1 and oc[i] == 50 and om[i] == 0) then already = false end
        end
        if already then
            notify("SRC", "values already look modded, off might not restore. rejoin to fully reset", 4)
        end
        ms = { G = G, C = C, M = M, og = og, oc = oc, om = om, on = false }
        _G[MSK] = ms
        msBusy = false
        notify("SRC", "minesweeper scan took " .. string.format("%.1f", os.clock() - o) .. "s", 3)
        if after then after() end
    end)
end

local function setMines(on)
    if on == msApplied and ms then return end
    if not ms then
        if not on then return end
        scanMines(function() msWrite(S.mines) end)
        return
    end
    msWrite(on)
    notify("SRC", "minesweeper " .. (on and "on" or "off"), 2)
end

S.mines = msApplied
task.spawn(function()
    while running do
        local ok, v = pcall(UI.GetValue, "tsc_mines")
        if ok and v ~= nil then S.mines = v end
        if not msBusy and (S.mines ~= msApplied or (S.mines and not ms)) then
            setMines(S.mines)
        end
        task.wait(0.25)
    end
end)

_G.TSC = { unload = function()
    running = false
    for _, o in ipairs(owned) do
        pcall(function() o.d.Visible = false end)
        STASH[o.kind] = STASH[o.kind] or {}
        local list = STASH[o.kind]
        list[#list + 1] = o.d
    end
    owned = {}
    restoreTurrets()
    restoreGates("lasers")
    restoreGates("teslas")
    restoreHazards()
    restoreAlarms()
    restoreVents()
    if ms and ms.on then pcall(msWrite, false) end
    pcall(UI.RemoveTab, "SRC")
    _G.TSC = nil
end }
